# A player's child's rite of passage at 14.
ek2lc_life.0002 = {
	type = character_event
	title = ek2lc_life.0002.t
	desc = ek2lc_life.0002.desc
	theme = family
	trigger = {
		is_ai = yes
		OR = {
			father ?= { is_ai = no }
			mother ?= { is_ai = no }
		}
	}
	left_portrait = {
		character = scope:ek2lc_parent
		animation = personality_honorable
	}
	right_portrait = {
		character = root
		animation = personality_bold
	}
	immediate = {
		if = {
			limit = { father ?= { is_ai = no } }
			father = { save_scope_as = ek2lc_parent }
		}
		else = {
			mother = { save_scope_as = ek2lc_parent }
		}
	}
	option = {
		name = ek2lc_life.0002.a
		add_character_modifier = { modifier = ek2lc_clan_mark_bear_modifier years = 10 }
		add_opinion = {
			target = scope:ek2lc_parent
			modifier = grateful_opinion
			opinion = 20
		}
		scope:ek2lc_parent = { add_prestige = minor_prestige_gain }
	}
	option = {
		name = ek2lc_life.0002.b
		add_character_modifier = { modifier = ek2lc_clan_mark_elk_modifier years = 10 }
	}
}

# The knot gives way at a feast.
ek2lc_life.0020 = {
	type = character_event
	title = ek2lc_life.0020.t
	desc = ek2lc_life.0020.desc
	theme = party
	trigger = {
		is_ai = no
		age >= 16
	}
	left_portrait = {
		character = root
		animation = shock
	}
	option = {
		name = ek2lc_life.0020.a
		add_prestige = minor_prestige_loss
		stress_impact = {
			shy = medium_stress_impact_gain
			gregarious = minor_stress_impact_loss
		}
	}
	option = {
		name = ek2lc_life.0020.b
		add_dread = 3
		stress_impact = {
			compassionate = minor_stress_impact_gain
			just = minor_stress_impact_gain
		}
	}
	option = {
		name = ek2lc_life.0020.c
		trigger = {
			OR = {
				diplomacy >= 12
				has_trait = gregarious
			}
		}
		add_character_modifier = { modifier = ek2lc_loincloth_fashion_modifier years = 5 }
	}
	after = {
		if = {
			limit = { NOT = { has_character_modifier = ek2lc_loincloth_fashion_modifier } }
			add_character_modifier = { modifier = ek2lc_tie_snapped_modifier years = 1 }
		}
	}
}

# A foreign guest is scandalised by court dress.
ek2lc_life.0040 = {
	type = character_event
	title = ek2lc_life.0040.t
	desc = ek2lc_life.0040.desc
	theme = diplomacy
	trigger = {
		is_ai = no
		is_ruler = yes
		any_courtier_or_guest = {
			is_adult = yes
			NOT = { ek2lc_culture_trigger = yes }
		}
	}
	left_portrait = {
		character = root
		animation = personality_bold
	}
	right_portrait = {
		character = scope:ek2lc_envoy
		animation = disgust
	}
	immediate = {
		random_courtier_or_guest = {
			limit = {
				is_adult = yes
				NOT = { ek2lc_culture_trigger = yes }
			}
			save_scope_as = ek2lc_envoy
		}
	}
	option = {
		name = ek2lc_life.0040.a
		add_character_modifier = { modifier = ek2lc_proud_dress_modifier years = 3 }
		scope:ek2lc_envoy = {
			add_opinion = {
				target = root
				modifier = insulted_opinion
				opinion = -10
			}
		}
	}
	option = {
		name = ek2lc_life.0040.b
		remove_short_term_gold = 10
		scope:ek2lc_envoy = {
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 15
			}
		}
	}
	option = {
		name = ek2lc_life.0040.c
		trigger = { diplomacy >= 10 }
		add_prestige = minor_prestige_gain
		scope:ek2lc_envoy = {
			add_opinion = {
				target = root
				modifier = respect_opinion
				opinion = 10
			}
		}
	}
}
