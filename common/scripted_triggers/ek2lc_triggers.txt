# EK2 Loincloths: who wears what. Only reads existing triggers; changes no gameplay.

# Adults in a war role while their realm is at war: knights, commanders, anyone in an army.
ek2lc_war_role_trigger = {
	exists = this
	age >= 14  # brief: teens 14+ in war roles get the war and armoured looks
	is_incapable = no
	is_imprisoned = no
	OR = {
		is_commanding_army = yes
		is_in_army = yes
		is_knight = yes
	}
	OR = {
		is_at_war = yes
		liege ?= { is_at_war = yes }
	}
}

# The best fighters in a war role get the armoured look instead: knights, or prowess 12 and up.
ek2lc_armour_role_trigger = {
	ek2lc_war_role_trigger = yes
	OR = {
		is_knight = yes
		prowess >= 12
	}
}

# Rulers and their spouses (adults): their royal look follows the higher of their two top title tiers.
ek2lc_royal_trigger = {
	exists = this  # child rulers too (brief); the modifiers leave out children under 4
	OR = {
		highest_held_title_tier >= tier_barony
		primary_spouse ?= { highest_held_title_tier >= tier_barony }
	}
}

# A ruler or spouse whose higher top title tier is from tier_$LOW$ to tier_$HIGH$.
ek2lc_royal_tier_trigger = {
	ek2lc_royal_trigger = yes
	OR = {
		highest_held_title_tier >= tier_$LOW$
		primary_spouse ?= { highest_held_title_tier >= tier_$LOW$ }
	}
	highest_held_title_tier <= tier_$HIGH$
	NOT = { primary_spouse ?= { highest_held_title_tier > tier_$HIGH$ } }
}

# Camp followers: people who travel with an army but don't fight. The courtiers of a landless
# adventurer's camp (not the adventurer), and anyone whose spouse is away with an army.
# Never a knight or anyone in a war role.
ek2lc_camp_follower_trigger = {
	exists = this
	is_imprisoned = no
	is_knight = no
	ek2lc_war_role_trigger = no
	OR = {
		AND = {
			is_ruler = no  # (a courtier reports the camp's government, so is_landless_adventurer can't tell them apart)
			liege ?= { is_landless_adventurer = yes }
		}
		any_spouse = { is_in_army = yes }
	}
}
