# EK2 Loincloths: order ceremonial regalia from the weavers (events/ek2lc_regalia_events.txt).
ek2lc_commission_regalia_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ek2lc_commission_regalia.dds"
	}
	desc = ek2lc_commission_regalia_decision_desc
	selection_tooltip = ek2lc_commission_regalia_decision_tooltip
	sort_order = 90
	cooldown = { years = 2 }

	is_shown = {
		has_game_rule = ek2lc_events_on
		is_adult = yes
		is_landed = yes
		ek2lc_loincloth_people_trigger = yes
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		custom_tooltip = {
			text = ek2lc_commission_pending_tt
			NOT = { has_character_flag = ek2lc_commission_pending }
		}
		gold >= ek2lc_commission_rough_cost
	}

	effect = {
		custom_tooltip = ek2lc_commission_regalia_decision_effect_tt
		trigger_event = ek2lc_regalia.0001
	}

	ai_check_interval = 120
	ai_potential = {
		highest_held_title_tier >= tier_county
		gold >= 300
		NOT = { ek2lc_owns_regalia_trigger = yes }
	}
	ai_will_do = {
		base = 10
	}
}

# A season when the whole court wears the white festival wrap (events: the wrap shows while the modifier lasts).
ek2lc_loincloth_festival_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/ek2lc_festival.dds"
	}
	desc = ek2lc_loincloth_festival_decision_desc
	selection_tooltip = ek2lc_loincloth_festival_decision_tooltip
	sort_order = 89
	cooldown = { years = 5 }
	is_shown = {
		has_game_rule = ek2lc_events_on
		is_ruler = yes
		is_landed = yes
		ek2lc_loincloth_people_trigger = yes
	}
	is_valid_showing_failures_only = {
		is_available_adult = yes
	}
	cost = {
		gold = 50
		prestige = 100
	}
	effect = {
		add_character_modifier = { modifier = ek2lc_loincloth_festival_modifier days = 90 }
		every_courtier = {
			limit = { ek2lc_loincloth_people_trigger = yes }
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 10
			}
		}
		every_vassal = {
			limit = { ek2lc_loincloth_people_trigger = yes }
			add_opinion = {
				target = root
				modifier = pleased_opinion
				opinion = 5
			}
		}
	}
	ai_check_interval = 120
	ai_potential = { gold >= 200 prestige >= 300 }
	ai_will_do = { base = 5 }
}
