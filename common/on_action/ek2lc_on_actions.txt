# EK2 Loincloths: hooks for the loincloth events (appended to vanilla's on_actions, nothing replaced).
on_birthday = {
	on_actions = { ek2lc_on_birthday }
}
ek2lc_on_birthday = {
	trigger = {
		has_game_rule = ek2lc_events_on
		age = 14
		ek2lc_loincloth_people_trigger = yes
		NOT = { has_trait = undead_character }
	}
	events = {
		ek2lc_life.0001  # a player's own rite
		ek2lc_life.0002  # a player's child's rite
	}
}

on_combat_end_winner = {
	on_actions = { ek2lc_on_combat_won }
}
ek2lc_on_combat_won = {
	effect = {  # counted whatever the events rule says: the cowrie war belt shows battles won
		every_side_commander = {
			limit = { ek2lc_loincloth_people_trigger = yes }
			ek2lc_count_battle_effect = yes
		}
		every_side_knight = {
			limit = { ek2lc_loincloth_people_trigger = yes }
			ek2lc_count_battle_effect = yes
		}
	}
}

yearly_playable_pulse = {
	on_actions = { ek2lc_yearly_pulse }
}
ek2lc_yearly_pulse = {
	trigger = {
		has_game_rule = ek2lc_events_on
		is_adult = yes
		ek2lc_loincloth_people_trigger = yes
		NOT = { has_trait = undead_character }
	}
	random_events = {
		900 = 0
		60 = ek2lc_regalia.0020  # a patron's gift of a Champion's Apron
		60 = ek2lc_regalia.0030  # moths in the regalia
		40 = ek2lc_regalia.0040  # wearing an inherited regalia at court
		50 = ek2lc_life.0020  # the knot gives way at a feast
		40 = ek2lc_life.0030  # the weavers' sumptuary petition
		40 = ek2lc_life.0040  # a scandalised envoy
		40 = ek2lc_life.0050  # Hist sap
		40 = ek2lc_life.0070  # Hircine's night
		40 = ek2lc_life.0080  # a cold snap
		40 = ek2lc_regalia.0050  # regalia stolen at a feast
		50 = ek2lc_regalia.0060  # a promising weaver at court
	}
}

# Close family (and spouses) of a loincloth people mourn a year: the dark mourning wrap.
on_death = {
	on_actions = { ek2lc_on_death }
}
ek2lc_on_death = {
	effect = {
		every_close_family_member = {
			limit = {
				is_alive = yes
				age >= 14
				ek2lc_loincloth_people_trigger = yes
			}
			add_character_modifier = { modifier = ek2lc_mourning_modifier years = 1 }
		}
		every_spouse = {
			limit = {
				is_alive = yes
				ek2lc_loincloth_people_trigger = yes
			}
			add_character_modifier = { modifier = ek2lc_mourning_modifier years = 1 }
		}
	}
}
