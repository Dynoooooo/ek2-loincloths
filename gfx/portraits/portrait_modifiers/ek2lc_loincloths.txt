# EK2 Loincloths: who wears a loincloth.
# Priority 101 applies this after every EK2 outfit group (EK2's highest is 100), so a
# loincloth wearer never gets an EK2 outfit, legwear or cloak on top of the loincloth.
# Every modifier sets all of our genes, so nothing is left over from another choice.
ek2lc_loincloth = {
	usage = game
	selection_behavior = weighted_random
	priority = 101

	ek2lc_none = {
		dna_modifiers = {
			accessory = {
				mode = add
				gene = gene_ek2lc_loincloth
				template = ek2lc_no_loincloth
				range = { 0 1 }
			}
			accessory = {
				mode = add
				gene = gene_ek2lc_top
				template = ek2lc_no_top
				range = { 0 1 }
			}
			accessory = {
				mode = add
				gene = gene_ek2lc_war
				template = ek2lc_no_war
				range = { 0 1 }
			}
		}
		weight = {
			base = 1
		}
	}
